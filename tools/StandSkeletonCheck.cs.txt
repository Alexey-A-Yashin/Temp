using System;
using System.IO;
using System.Text;
using System.Collections.Generic;
using ColoringBook.Domain.Level;
using ColoringBook.Infrastructure.Generation.Plants;

// СКЕЛЕТ БЕЗ ЛИСТВЫ: то, что видно на зимнем дереве. Меряем углы отхождения ветвей,
// отношение диаметров ветви к стволу и число ветвей первого порядка.
public static class Program
{
    public static void Main(string[] a)
    {
        string dir = a.Length > 0 ? a[0] : "/tmp/skel";
        Directory.CreateDirectory(dir);
        PlantGenerator.Detail = PlantGenerator.FoliageDetail.Compact;
        PlantGenerator.QualityLevel = PlantGenerator.GameQEffTarget;
        PlantGenerator.AgeLevel = PlantGenerator.GameAgeMax;

        Console.WriteLine($"{"порода",-12}{"угол ветви",12}{"натура",12}{"ветвь/ствол",13}{"натура",10}");
        Console.WriteLine(new string('-', 62));
        foreach (var kind in new[] { VegetationKind.Fir, VegetationKind.Pine, VegetationKind.Deciduous })
        for (int sp = 0; sp < PlantSpecies.Count(kind); sp++)
        {
            var d = PlantGenerator.Generate(PlantGenome.Create(kind, 4242 + sp * 7919, sp));
            // Скелет: только древесные карточки. Извлекаем сегменты как пары вершин
            // соседних колец — приблизительно, по направлению рёбер.
            var dirs = new List<(float ang, float len)>();
            float h = 0f;
            for (int i = 0; i < d.VertexCount; i++) h = MathF.Max(h, d.Positions[i * 3 + 1]);

            // Углы: берём древесные квады и считаем наклон их длинной оси к вертикали.
            int quads = d.Indices.Count / 6;
            var angles = new List<float>();
            for (int q = 0; q < quads; q++)
            {
                int i0 = d.Indices[q * 6], i1 = d.Indices[q * 6 + 1], i3 = d.Indices[q * 6 + 5];
                if (FoliageAtlas.IsFoliageUv(d.UVs[i0 * 2], d.UVs[i0 * 2 + 1])) continue;
                float dx = d.Positions[i1 * 3] - d.Positions[i0 * 3];
                float dy = d.Positions[i1 * 3 + 1] - d.Positions[i0 * 3 + 1];
                float dz = d.Positions[i1 * 3 + 2] - d.Positions[i0 * 3 + 2];
                float ln = MathF.Sqrt(dx * dx + dy * dy + dz * dz);
                if (ln < 1e-4f) continue;
                // Ось сегмента — это вторая сторона квада (i0->i3 идёт вдоль ветви)
                float ex = d.Positions[i3 * 3] - d.Positions[i0 * 3];
                float ey = d.Positions[i3 * 3 + 1] - d.Positions[i0 * 3 + 1];
                float ez = d.Positions[i3 * 3 + 2] - d.Positions[i0 * 3 + 2];
                float el = MathF.Sqrt(ex * ex + ey * ey + ez * ez);
                if (el < 1e-4f) continue;
                float cosv = MathF.Abs(ey / el);
                // угол к вертикали: 0 = вертикально, 90 = горизонтально
                angles.Add(MathF.Acos(Math.Clamp(cosv, 0f, 1f)) * 180f / MathF.PI);
            }
            angles.Sort();
            float med = angles.Count > 0 ? angles[angles.Count / 2] : 0;
            string n = PlantSpecies.Get(kind, sp).Name;
            string natAng = kind == VegetationKind.Deciduous ? "40–70°" : "60–85°";
            Console.WriteLine($"{n,-12}{med,11:F0}°{natAng,12}{"—",13}{"0.3–0.6",10}");

            // Выгружаем ТОЛЬКО древесину — зимний силуэт
            var sb = new StringBuilder();
            var map = new Dictionary<int,int>();
            var keep = new List<int>();
            for (int q = 0; q < quads; q++)
            {
                int i0 = d.Indices[q * 6];
                if (FoliageAtlas.IsFoliageUv(d.UVs[i0 * 2], d.UVs[i0 * 2 + 1])) continue;
                for (int t = 0; t < 6; t++) keep.Add(d.Indices[q * 6 + t]);
            }
            var uniq = new List<int>(new HashSet<int>(keep));
            for (int i = 0; i < uniq.Count; i++) map[uniq[i]] = i;
            foreach (int v in uniq)
                sb.AppendLine($"v {d.Positions[v*3]:F5} {d.Positions[v*3+1]:F5} {d.Positions[v*3+2]:F5} {d.UVs[v*2]:F5} {d.UVs[v*2+1]:F5} {d.Colors[v*4]:F4} {d.Colors[v*4+1]:F4} {d.Colors[v*4+2]:F4}");
            for (int t = 0; t + 2 < keep.Count; t += 3)
                sb.AppendLine($"f {map[keep[t]]} {map[keep[t+1]]} {map[keep[t+2]]}");
            File.WriteAllText(Path.Combine(dir, n + "_skel.mesh"), sb.ToString());
        }
    }
}
