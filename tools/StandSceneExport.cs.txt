using System;
using System.Globalization;
using System.IO;
using System.Text;
using ColoringBook.Domain.Common;
using ColoringBook.Domain.Level;
using ColoringBook.Domain.Level.Season;
using ColoringBook.Infrastructure.Generation.Mapgen;
using ColoringBook.Infrastructure.Generation.Plants;

// Выгрузка сцены для внешней проверки обхода уровня.
public static class Program
{
    static string F(float v) => v.ToString("F4", CultureInfo.InvariantCulture);
    public static void Main(string[] args)
    {
        string dir = args.Length > 0 ? args[0] : "/tmp/scene";
        Directory.CreateDirectory(dir);
        var gen = new MapgenLevelGenerator(new GraphicsSettings(GraphicsQuality.Medium));
        var level = gen.Generate(new LevelDescriptor(Guid.NewGuid(), 12345,
            SeasonMode.Fixed, SeasonType.Summer, DateTime.UtcNow));
        var sb = new StringBuilder();
        foreach (var r in level.Regions)
            sb.AppendLine($"{F(r.Position.X)} {F(r.Position.Y)} {F(r.Elevation)} {(int)r.Biome} {(r.IsRiver ? 1 : 0)} {F(r.WaterLevel)}");
        File.WriteAllText(Path.Combine(dir, "regions.txt"), sb.ToString());
        sb.Clear();
        foreach (var p in level.Vegetation)
            sb.AppendLine($"{F(p.X)} {F(p.Z)} {(int)p.Kind} {p.VariantIndex} {F(p.Scale)}");
        File.WriteAllText(Path.Combine(dir, "plants.txt"), sb.ToString());
        // высоты прототипов
        PlantCatalog.EnsureGamePool();
        sb.Clear();
        foreach (VegetationKind k in Enum.GetValues(typeof(VegetationKind)))
            for (int v = 0; v < PlantCatalog.VariantsPerKind; v++)
            {
                var m = PlantCatalog.Instance.Get(k, v);
                float h = 0f;
                for (int i = 0; i < m.VertexCount; i++) h = MathF.Max(h, m.Positions[i*3+1]);
                sb.AppendLine($"{(int)k} {v} {F(h)} {F(m.LocalHeight)} {m.Indices.Count/3}");
            }
        File.WriteAllText(Path.Combine(dir, "protos.txt"), sb.ToString());
        Console.WriteLine($"регионов {level.Regions.Count}, растений {level.Vegetation.Count}");
    }
}
